## استفاده

```php
use Tavaan\Engine\View\ViewFactory;
use Tavaan\Engine\View\View;
use Tavaan\Engine\View\Escaper;

// از Factory
$factory = new ViewFactory(viewPath: 'views');
$factory->share('app_name', 'Tavaan');
$view = $factory->make('user.profile', ['user' => $user]);
echo $view->render();

// مستقیم
$view = new View('Hello, {{ $name }}!', ['name' => 'Ali']);
echo $view->render();
echo $view;  // __toString()

// با
$view = (new View('...'))->with('name', 'Ali')->with('age', 25);

// Escape
Escaper::html('<script>');     // '&lt;script&gt;'
Escaper::attr('value');        // attribute-safe
Escaper::js($data);             // JS-safe JSON
Escaper::url('hello world');   // 'hello%20world'
```
